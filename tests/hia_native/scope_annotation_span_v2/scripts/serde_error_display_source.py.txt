def publish(o:mutbuf, w:mutbuf) -> u64:
    code = pget(w, 48)
    if code == 0:
        clear(o, 0, 64)
        put(o, 0, RESPONSE)
        return 64
    at = pget(w, 64)
    end = pget(w, 56)
    line = 1
    column = 0
    while at < end:
        ch = byte(w, at)
        if ch == 10:
            line = line + 1
            column = 0
        else:
            column = column + 1
        at = at + 1
    count = literal_size(code)
    if code == 105 or code == 106:
        count = count + pget(w, 192) + literal_size(107)
    count = count + literal_size(200) + decimal_size(line) + literal_size(201) + decimal_size(column)
    if count > length(o) - 64:
        pput(w,200,count+64)
        return CAPACITY
    clear(o, 0, 64)
    put(o, 0, RESPONSE)
    put(o, 8, 1)
    put(o, 16, pget(w, 72))
    put(o, 24, line)
    put(o, 32, column)
    put(o, 40, count)
    at = literal(o, 64, code)
    if code == 105 or code == 106:
        copy_range(o, w, at, NUMBER_TEXT, pget(w, 192))
        at = at + pget(w, 192)
        at = literal(o, at, 107)
    at = literal(o, at, 200)
    at = decimal(o, at, line)
    at = literal(o, at, 201)
    at = decimal(o, at, column)
    return at

def decimal_size(value:u64) -> u64:
    digits = 1
    power = 10
    while digits < 20 and value >= power:
        digits = digits + 1
        if digits < 20:
            power = power * 10
    return digits

def decimal(o:mutbuf, at:u64, value:u64) -> u64:
    digits = decimal_size(value)
    power = 1
    remaining = digits
    while remaining > 1:
        power = power * 10
        remaining = remaining - 1
    while digits > 0:
        digit = 0
        while value >= power:
            value = value - power
            digit = digit + 1
        store(o, at, 48 + digit)
        at = at + 1
        digits = digits - 1
        power = 1
        remaining = digits
        while remaining > 1:
            power = power * 10
            remaining = remaining - 1
    return at
