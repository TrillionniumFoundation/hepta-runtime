def entry(s:readbuf, o:mutbuf, w:mutbuf) -> u32:
    if length(s) < 32 or read64(s, 0) != REQUEST:
        return INVALID
    operation = read64(s, 8)
    raw_size = read64(s, 16)
    filename_size = read64(s, 24)
    if operation != 1 and operation != 2:
        return INVALID
    if filename_size != choose(operation == 1, 0, 69):
        return INVALID
    if filename_size > length(s) - 32 or raw_size != length(s) - 32 - filename_size:
        return INVALID
    if length(o) < 32 or length(w) < 16384:
        return CAPACITY
    if operation == 2 and valid_filename(s) == 0:
        return error(o, 4)
    if raw_size > 2048:
        return error(o, 3)
    fill_range(w, 0, 4096, 0)
    start = 32 + filename_size
    end = parse_root(s, w, whitespace(s, start))
    if end >= CAPACITY:
        return error(o, 1)
    if whitespace(s, end) != length(s):
        return error(o, 1)
    if operation == 2:
        if valid_identifier(s, w, 0) == 0 or valid_identifier(s, w, 1) == 0 or valid_tool(s, w) == 0:
            return error(o, 2)
        i = 0
        at = get(w, 256)
        while i < 64:
            decoded_value = decoded(s, at)
            if (decoded_value & 4294967295) != byte(s, 32 + i):
                return error(o, 2)
            at = decoded_value >> 32
            i = i + 1
        if byte(s, 96) != 46 or byte(s, 97) != 106 or byte(s, 98) != 115 or byte(s, 99) != 111 or byte(s, 100) != 110:
            return error(o, 2)
    required = 128 + get(w, 272) + get(w, 296) + get(w, 320)
    if length(o) < required:
        return CAPACITY
    fill_range(o, 0, 128, 0)
    put(o, 0, RESPONSE)
    put(o, 16, get(w, 64))
    put(o, 24, get(w, 72))
    put(o, 32, get(w, 80))
    put(o, 40, get(w, 88))
    put(o, 48, get(w, 96))
    put(o, 56, required)
    offset = 128
    i = 0
    while i < 3:
        put(o, 64 + i * 16, offset)
        put(o, 72 + i * 16, get(w, 272 + i * 24))
        offset = copy_string(s, o, get(w, 256 + i * 24), offset)
        i = i + 1
    return required

def error(o:mutbuf, diagnostic:u64) -> u64:
    fill_range(o, 0, 32, 0)
    put(o, 0, RESPONSE)
    put(o, 8, diagnostic)
    return 32

def parse_root(s:readbuf, w:mutbuf, at:u64) -> u64:
    if at >= length(s):
        return INVALID
    if byte(s, at) == 91:
        i = 0
        at = whitespace(s, at + 1)
        while i < 6:
            at = parse_field(s, w, at, i + 1)
            if at >= CAPACITY:
                return at
            at = whitespace(s, at)
            if at >= length(s):
                return INVALID
            if i < 5:
                if byte(s, at) != 44:
                    return INVALID
                at = whitespace(s, at + 1)
            else:
                if byte(s, at) != 93:
                    return INVALID
                return at + 1
            i = i + 1
        return INVALID
    if byte(s, at) != 123:
        return INVALID
    at = whitespace(s, at + 1)
    mask = 0
    running = 1
    while running == 1:
        if at >= length(s) or byte(s, at) != 34:
            return INVALID
        end = string_info(s, w, at)
        if end >= CAPACITY:
            return end
        field = identify(s, at + 1, get(w, 24), 1, 6)
        if field == 0:
            return INVALID
        flag = 1 << (field - 1)
        if (mask & flag) != 0:
            return INVALID
        mask = mask | flag
        at = whitespace(s, end)
        if at >= length(s) or byte(s, at) != 58:
            return INVALID
        at = parse_field(s, w, whitespace(s, at + 1), field)
        if at >= CAPACITY:
            return at
        at = whitespace(s, at)
        if at >= length(s):
            return INVALID
        if byte(s, at) == 125:
            if (mask & 31) != 31:
                return INVALID
            return at + 1
        if byte(s, at) != 44:
            return INVALID
        at = whitespace(s, at + 1)
    return INVALID

def parse_field(s:readbuf, w:mutbuf, at:u64, field:u64) -> u64:
    if at >= length(s):
        return INVALID
    if field <= 3:
        if byte(s, at) != 34:
            return INVALID
        end = string_info(s, w, at)
        if end >= CAPACITY:
            return end
        slot = 256 + (field - 1) * 24
        put(w, slot, at + 1)
        put(w, slot + 8, end)
        put(w, slot + 16, get(w, 24))
        return end
    if field == 4:
        return parse_state(s, w, at)
    if field == 5:
        end = unsigned(s, w, at, 128)
        if end >= CAPACITY:
            return end
        put(w, 72, get(w, 8))
        put(w, 80, get(w, 16))
        return end
    if byte(s, at) == 110:
        end = null(s, at)
        if end >= CAPACITY:
            return end
        put(w, 88, 0)
        put(w, 96, 0)
        return end
    end = unsigned(s, w, at, 64)
    if end >= CAPACITY:
        return end
    put(w, 88, 1)
    put(w, 96, get(w, 8))
    return end

def parse_state(s:readbuf, w:mutbuf, at:u64) -> u64:
    tagged = byte(s, at) == 123
    if tagged:
        at = whitespace(s, at + 1)
    if at >= length(s) or byte(s, at) != 34:
        return INVALID
    end = string_info(s, w, at)
    if end >= CAPACITY:
        return end
    code = identify(s, at + 1, get(w, 24), 7, 9)
    if code == 0:
        return INVALID
    put(w, 64, code - 7)
    if tagged == False:
        return end
    at = whitespace(s, end)
    if at >= length(s) or byte(s, at) != 58:
        return INVALID
    at = null(s, whitespace(s, at + 1))
    if at >= CAPACITY:
        return at
    at = whitespace(s, at)
    if at >= length(s) or byte(s, at) != 125:
        return INVALID
    return at + 1

def identify(s:readbuf, start:u64, size:u64, first:u64, last:u64) -> u64:
    code = first
    while code <= last:
        if name_length(code) == size:
            at = start
            i = 0
            equal = 1
            while i < size and equal == 1:
                value = decoded(s, at)
                if (value & 4294967295) != name_byte(code, i):
                    equal = 0
                at = value >> 32
                i = i + 1
            if equal == 1:
                return code
        code = code + 1
    return 0

def valid_identifier(s:readbuf, w:mutbuf, index:u64) -> u64:
    if get(w, 272 + index * 24) != 64:
        return 0
    at = get(w, 256 + index * 24)
    i = 0
    while i < 64:
        value = decoded(s, at)
        ch = value & 4294967295
        if (ch < 48 or ch > 57) and (ch < 97 or ch > 102):
            return 0
        at = value >> 32
        i = i + 1
    return 1

def valid_tool(s:readbuf, w:mutbuf) -> u64:
    size = get(w, 320)
    if size == 0 or size > 64:
        return 0
    at = get(w, 304)
    i = 0
    while i < size:
        value = decoded(s, at)
        ch = value & 4294967295
        if (ch < 97 or ch > 122) and ch != 95:
            return 0
        at = value >> 32
        i = i + 1
    return 1

def valid_filename(s:readbuf) -> u64:
    i = 0
    while i < 64:
        ch = byte(s, 32 + i)
        if (ch < 48 or ch > 57) and (ch < 97 or ch > 102):
            return 0
        i = i + 1
    if byte(s, 96) != 46 or byte(s, 97) != 106 or byte(s, 98) != 115 or byte(s, 99) != 111 or byte(s, 100) != 110:
        return 0
    return 1
