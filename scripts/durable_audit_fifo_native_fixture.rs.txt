use std::collections::VecDeque;
use std::sync::{Arc,Mutex,OnceLock};
use std::time::{Instant,Duration};
use std::io::Write;
use std::os::fd::{FromRawFd,AsRawFd};
use sha2::{Sha256,Digest};
type Entry=unsafe extern "C" fn(*const u8,u64,*mut u8,u64,*mut u8,u64)->u64;
struct Code{entry:Entry}
unsafe impl Send for Code{}
unsafe impl Sync for Code{}
impl Code{
 fn get()->&'static Self{
  static CODE:OnceLock<Code>=OnceLock::new();
  CODE.get_or_init(||{
   let path=std::env::var("HIA_FIFO_FIXTURE_IMAGE").expect("explicit private image");
   let expected=std::env::var("HIA_FIFO_FIXTURE_IMAGE_SHA256").expect("explicit complete image SHA");
   let bytes=std::fs::read(path).unwrap();
   assert_eq!(Sha256::digest(&bytes).iter().map(|b|format!("{b:02x}")).collect::<String>(),expected);
   let name=std::ffi::CString::new("audit-fifo-qualified").unwrap();
   let fd=unsafe{libc::memfd_create(name.as_ptr(),libc::MFD_CLOEXEC|libc::MFD_ALLOW_SEALING)};assert!(fd>=0);
   let mut file=unsafe{std::fs::File::from_raw_fd(fd)};file.write_all(&bytes).unwrap();
   let seals=libc::F_SEAL_WRITE|libc::F_SEAL_SHRINK|libc::F_SEAL_GROW|libc::F_SEAL_SEAL;
   assert_eq!(unsafe{libc::fcntl(file.as_raw_fd(),libc::F_ADD_SEALS,seals)},0);
   assert_eq!(unsafe{libc::fcntl(file.as_raw_fd(),libc::F_GET_SEALS)},seals);
   let address=unsafe{libc::mmap(std::ptr::null_mut(),bytes.len(),libc::PROT_READ|libc::PROT_EXEC,libc::MAP_PRIVATE,file.as_raw_fd(),0)};
   assert_ne!(address,libc::MAP_FAILED);
   Self{entry:unsafe{std::mem::transmute::<*mut libc::c_void,Entry>(address)}}
  })
 }
}
fn clock_origin()->Instant{static ORIGIN:OnceLock<Instant>=OnceLock::new();*ORIGIN.get_or_init(Instant::now)}
fn tick(value:Instant)->u64{u64::try_from(value.saturating_duration_since(clock_origin()).as_nanos()).unwrap()}
fn get(raw:&[u8],at:usize)->u64{u64::from_le_bytes(raw[at..at+8].try_into().unwrap())}
fn put(raw:&mut[u8],at:usize,value:u64){raw[at..at+8].copy_from_slice(&value.to_le_bytes())}
#[derive(Clone,Copy,PartialEq,Eq)]enum AuditStoreAccess{Exclusive,Shared}
#[derive(Clone,Copy,Debug,PartialEq,Eq)]enum StoreLockFailure{Deadline,Unavailable}
struct AuditStoreQueue{
 next_ticket:u64,active:bool,readers:usize,reader_poisoned:bool,
 waiting:VecDeque<(u64,AuditStoreAccess)>,state:Vec<u8>
}
impl Default for AuditStoreQueue{
 fn default()->Self{
  let _=clock_origin();let mut result=Self{next_ticket:0,active:false,readers:0,reader_poisoned:false,waiting:VecDeque::new(),state:vec![0;128]};
  assert_eq!(result.call_at(0,None,0,0,0)[1],0);result
 }
}
impl AuditStoreQueue{
 fn publish_observation(&mut self){
  self.next_ticket=get(&self.state,8);self.active=get(&self.state,16)==1;self.readers=get(&self.state,32)as usize;self.reader_poisoned=get(&self.state,40)==1;
  self.waiting.clear();
  for i in 0..get(&self.state,48)as usize{let at=128+i*64;if get(&self.state,at+16)==1{self.waiting.push_back((get(&self.state,at),if get(&self.state,at+8)==1{AuditStoreAccess::Exclusive}else{AuditStoreAccess::Shared}));}}
 }
 fn call_at(&mut self,op:u64,deadline:Option<Instant>,ticket:u64,flags:u64,now:u64)->[u64;8]{
  let mut input=[0u8;64];input[..8].copy_from_slice(b"HRAFL001");
  for(at,value)in[(8,op),(16,now),(24,u64::from(deadline.is_some())),(32,deadline.map(tick).unwrap_or(0)),(40,ticket),(48,flags)]{put(&mut input,at,value)}
  loop{
   let original=input;let state_before=self.state.clone();let mut output=[0xa5u8;96];let input_ptr=input.as_ptr();
   // The 16-byte output canaries are outside the exact declared 64-byte span.
   let code=unsafe{(Code::get().entry)(input_ptr,64,output.as_mut_ptr().add(16),64,self.state.as_mut_ptr(),self.state.len()as u64)};
   assert_eq!(input,original);assert_eq!(&output[..16],&[0xa5;16]);assert_eq!(&output[80..],&[0xa5;16]);
   assert_eq!(code,64,"native private FIFO protocol refusal");
   let reply=std::array::from_fn(|i|get(&output[16..80],i*8));
   assert_eq!(&output[16..24],b"HRAFO001");assert_eq!(&reply[5..],&[0,0,0]);
   if reply[1]==7{
    assert_eq!(self.state,state_before,"growth must not consume a ticket or change ownership");
    let required=usize::try_from(reply[3]).unwrap();assert!(required>self.state.len());
    let capacity=required.max(self.state.len().checked_mul(2).unwrap());self.state.resize(capacity,0);continue
   }
   self.publish_observation();return reply
  }
 }
 fn step(&mut self,op:u64,deadline:Option<Instant>,ticket:u64,flags:u64)->[u64;8]{self.call_at(op,deadline,ticket,flags,tick(Instant::now()))}
}
#[derive(Default)]struct AuditStoreLock{inner:Mutex<()>,queue:Mutex<AuditStoreQueue>,changed:std::sync::Condvar}
struct AuditStoreGuard<'a>{lock:&'a AuditStoreLock,inner:Option<std::sync::MutexGuard<'a,()>>,ticket:u64,deadline:Option<Instant>,active:bool}
struct AuditSharedGuard<'a>{lock:&'a AuditStoreLock,deadline:Instant,ticket:u64,active:bool,_not_send:std::marker::PhantomData<std::sync::MutexGuard<'a,()>>}
fn failure(code:u64)->StoreLockFailure{match code{4=>StoreLockFailure::Deadline,5=>StoreLockFailure::Unavailable,_=>panic!("unexpected native status {code}")}}
fn preflight(deadline:Option<Instant>)->Result<(),StoreLockFailure>{
 let mut state=AuditStoreQueue::default();let result=state.step(10,deadline,0,0);if result[1]==0{Ok(())}else{Err(failure(result[1]))}
}
impl AuditStoreLock{
 fn flags(&self)->u64{u64::from(self.inner.is_poisoned())*2}
 fn resume(&self,op:u64,deadline:Option<Instant>,ticket:u64,extra:u64)->[u64;8]{
  let (mut queue,poison)=match self.queue.lock(){Ok(q)=>(q,0),Err(e)=>(e.into_inner(),1)};
  let result=queue.step(op,deadline,ticket,self.flags()|poison|extra);
  if result[4]==1{self.changed.notify_all()}result
 }
 fn reserve(&self,deadline:Option<Instant>,access:u64)->Result<u64,StoreLockFailure>{
  preflight(deadline)?;
  let mut queue=self.queue.lock().map_err(|_|StoreLockFailure::Unavailable)?;
  let first=queue.step(access,deadline,0,self.flags());
  if first[1]!=1{return Err(failure(first[1]))}
  let ticket=first[2];
  loop{
   let result=queue.step(3,deadline,ticket,self.flags());
   if result[4]==1{self.changed.notify_all()}
   match result[1]{2=>return Ok(ticket),1=>{},code=>return Err(failure(code))}
   let waited=match deadline{
    Some(limit)=>self.changed.wait_timeout(queue,limit.saturating_duration_since(Instant::now())).map(|(queue,_)|queue).map_err(|e|e.into_inner().0),
    None=>self.changed.wait(queue).map_err(|e|e.into_inner())
   };
   queue=match waited{Ok(q)=>q,Err(mut q)=>{let result=q.step(9,deadline,ticket,self.flags()|1);assert_eq!(result[1],5);self.changed.notify_all();return Err(StoreLockFailure::Unavailable)}};
  }
 }
 fn acquire(&self,deadline:Option<Instant>)->Result<AuditStoreGuard<'_>,StoreLockFailure>{
  let ticket=self.reserve(deadline,1)?;
  let mut guard=AuditStoreGuard{lock:self,inner:None,ticket,deadline,active:true};
  let before=self.resume(4,deadline,ticket,0);
  if before[1]!=8{return Err(failure(before[1]))}
  let flags=match self.inner.try_lock(){Ok(inner)=>{guard.inner=Some(inner);0},Err(std::sync::TryLockError::Poisoned(e))=>{drop(e.into_inner());2},Err(std::sync::TryLockError::WouldBlock)=>4};
  let after=self.resume(5,deadline,ticket,flags);
  if after[1]!=3{return Err(failure(after[1]))}
  Ok(guard)
 }
 fn acquire_shared(&self,deadline:Instant)->Result<AuditSharedGuard<'_>,StoreLockFailure>{
  let ticket=self.reserve(Some(deadline),2)?;
  let guard=AuditSharedGuard{lock:self,deadline,ticket,active:true,_not_send:std::marker::PhantomData};
  let result=self.resume(4,Some(deadline),ticket,0);
  if result[1]!=3{return Err(failure(result[1]))}Ok(guard)
 }
}
impl Drop for AuditStoreGuard<'_>{
 fn drop(&mut self){if !self.active{return}drop(self.inner.take());let result=self.lock.resume(6,self.deadline,self.ticket,0);assert_eq!(result[1],0);self.active=false;}
}
impl AuditSharedGuard<'_>{
 fn finish(mut self)->Result<(),StoreLockFailure>{
  let result=self.lock.resume(7,Some(self.deadline),self.ticket,0);self.active=false;
  if result[1]==0{Ok(())}else{Err(failure(result[1]))}
 }
}
impl Drop for AuditSharedGuard<'_>{
 fn drop(&mut self){if !self.active{return}let result=self.lock.resume(8,Some(self.deadline),self.ticket,u64::from(std::thread::panicking())*8);assert_eq!(result[1],0);self.active=false;}
}
