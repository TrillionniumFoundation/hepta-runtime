def entry(s:readbuf, o:mutbuf, w:mutbuf) -> u32:
    if length(s) < 32 or read64(s, 0) != REQUEST:
        return INVALID
    operation = read64(s, 8)
    if operation < 1 or operation > 3 or read64(s, 16) != length(s) - 32 or read64(s, 24) != 0:
        return INVALID
    if length(o) < 160 or length(w) < length(s) or length(w) - length(s) < 1024:
        return CAPACITY
    if get(w, 0) == 0:
        if zeros(w, 0, length(w)) == 0:
            return INVALID
        copy_range(w, s, 1024, 0, length(s))
        put(w, 16, length(s))
        put(w, 24, operation)
        put(w, 32, length(o))
        put(w, 40, length(w))
        put(w, 232, STATE)
        store(w, 768, 47)
        store(w, 800, 114)
        store(w, 801, 101)
        store(w, 802, 113)
        store(w, 803, 117)
        store(w, 804, 101)
        store(w, 805, 115)
        store(w, 806, 116)
        store(w, 807, 45)
        store(w, 808, 97)
        store(w, 809, 117)
        store(w, 810, 100)
        store(w, 811, 105)
        store(w, 812, 116)
        if utf8(s, 32, length(s) - 32) == 0:
            fail(w, 10, 0)
        else:
            if valid_root(s) == 0:
                fail(w, 11, 0)
            else:
                put(w, 0, 10)
    else:
        if get(w, 232) != STATE or get(w, 0) == 999 or get(w, 224) != 1:
            return INVALID
        if get(w, 16) != length(s) or get(w, 32) != length(o) or get(w, 40) != length(w):
            return INVALID
        if equal_inputs(s, w, 1024) == 0 or ack_valid(w) == 0:
            return INVALID
        put(w, 224, 0)
    while True:
        phase = get(w, 0)
        result = 4294967295
        if phase == 10 or phase == 11 or phase == 20 or phase == 21 or phase == 22 or phase == 23 or phase == 24:
            result = walk_root(w)
        if phase >= 60 and phase <= 69:
            result = open_store(w)
        if phase == 800 or phase == 900:
            result = finish(o, w)
        if result != 0:
            return result
        if get(w, 0) == phase:
            return INVALID
    return INVALID

def valid_root(s:readbuf) -> u64:
    if length(s) == 32 or byte(s, 32) != 47:
        return 0
    at = 32
    while at < length(s):
        while at < length(s) and byte(s, at) == 47:
            at = at + 1
        start = at
        while at < length(s) and byte(s, at) != 47:
            at = at + 1
        n = at - start
        if n == 1 and byte(s, start) == 46 or n == 2 and byte(s, start) == 46 and byte(s, start + 1) == 46:
            return 0
    return 1

def walk_root(w:mutbuf) -> u64:
    phase = get(w, 0)
    result = get(w, 328)
    error_number = get(w, 336)
    if phase == 10:
        return action(w, 11, 1, 768, 1, PATHFLAGS, 0, 11)
    if phase == 11:
        if error_number != 0:
            fail(w, 23, error_number)
        else:
            own(w, 1, 1)
            put(w, 112, 1)
            put(w, 0, 20)
    if phase == 20:
        at = get(w, 64)
        n = get(w, 16) - 32
        while at < n and byte(w, 1056 + at) == 47:
            at = at + 1
        start = at
        while at < n and byte(w, 1056 + at) != 47:
            at = at + 1
        put(w, 64, at)
        put(w, 80, 1056 + start)
        put(w, 88, at - start)
        if start == n:
            put(w, 0, 60)
        else:
            put(w, 0, 21)
    if phase == 21:
        return action(w, 2, get(w, 112), 512, 88, 0, 0, 22)
    if phase == 22:
        if error_number != 0:
            fail(w, 23, error_number)
        else:
            if (get(w, 512) & 73) == 0:
                fail(w, 12, 0)
            else:
                if contains_zero(w, get(w, 80), get(w, 88)) == 1:
                    fail(w, 13, 0)
                else:
                    at = get(w, 64)
                    n = get(w, 16) - 32
                    while at < n and byte(w, 1056 + at) == 47:
                        at = at + 1
                    flags = PATHFLAGS
                    if at == n:
                        flags = DIRFLAGS
                    target = 3 - get(w, 112)
                    put(w, 96, target)
                    return action(w, 1, target, get(w, 80), get(w, 88), flags, get(w, 112), 23)
    if phase == 23:
        if error_number != 0:
            fail(w, 23, error_number)
        else:
            own(w, get(w, 96), 1)
            return close_to(w, get(w, 112), 24)
    if phase == 24:
        put(w, 112, get(w, 96))
        put(w, 0, 20)
    return 0

def open_store(w:mutbuf) -> u64:
    phase = get(w, 0)
    result = get(w, 328)
    error_number = get(w, 336)
    if phase == 60:
        return action(w, 2, get(w, 112), 512, 88, 0, 0, 61)
    if phase == 61:
        if error_number != 0:
            fail(w, 23, error_number)
        else:
            if private_directory(w) == 0:
                fail(w, 14, 0)
            else:
                if get(w, 24) == 1:
                    copy_range(w, w, 640, 512, 88)
                    put(w, 120, get(w, 112))
                    put(w, 0, 900)
                else:
                    if get(w, 24) == 2:
                        return action(w, 12, get(w, 112), 800, 13, 448, 0, 62)
                    else:
                        put(w, 0, 66)
    if phase == 62:
        if error_number != 0 and error_number != 17:
            fail(w, 23, error_number)
        else:
            if error_number == 0:
                put(w, 0, 64)
            else:
                put(w, 0, 66)
    if phase == 64:
        return action(w, 5, get(w, 112), 0, 0, 0, 0, 65)
    if phase == 65:
        if error_number == 4:
            put(w, 0, 64)
        else:
            if error_number != 0:
                fail(w, 23, error_number)
            else:
                put(w, 0, 66)
    if phase == 66:
        return action(w, 1, 3, 800, 13, DIRFLAGS, get(w, 112), 67)
    if phase == 67:
        if get(w, 24) == 3 and error_number == 2:
            put(w, 0, 900)
        else:
            if error_number != 0:
                fail(w, 23, error_number)
            else:
                own(w, 3, 1)
                put(w, 0, 68)
    if phase == 68:
        return action(w, 2, 3, 512, 88, 0, 0, 69)
    if phase == 69:
        if error_number != 0:
            fail(w, 23, error_number)
        else:
            if private_directory(w) == 0:
                fail(w, 14, 0)
            else:
                copy_range(w, w, 640, 512, 88)
                put(w, 120, 3)
                put(w, 0, 900)
    return 0

def finish(o:mutbuf, w:mutbuf) -> u64:
    phase = get(w, 0)
    if phase == 800:
        own(w, get(w, 216), 0)
        put(w, 0, get(w, 208))
    if phase == 900:
        exported = get(w, 120)
        slot = 1
        while slot <= 3:
            if get(w, 128 + (slot - 1) * 8) == 1 and slot != exported:
                return close_to(w, slot, 900)
            slot = slot + 1
        fill_range(o, 0, 160, 0)
        put(o, 0, RESPONSE)
        put(o, 8, get(w, 48))
        put(o, 16, get(w, 56))
        put(o, 24, exported)
        put(o, 32, 160)
        put(o, 40, get(w, 24))
        if exported != 0:
            copy_range(o, w, 64, 640, 88)
            own(w, exported, 0)
        put(w, 0, 999)
        return 160
    return 0

def action(w:mutbuf, kind:u64, slot:u64, span:u64, count:u64, argument:u64, capacity:u64, phase:u64) -> u64:
    sequence = get(w, 8)
    if sequence == MAXWORD:
        return INVALID
    fill_range(w, 256, 96, 0)
    put(w, 256, ACTION)
    put(w, 264, sequence + 1)
    put(w, 272, kind)
    put(w, 280, slot)
    put(w, 288, span)
    put(w, 296, count)
    put(w, 304, argument)
    put(w, 312, capacity)
    copy_range(w, w, 384, 256, 64)
    put(w, 8, sequence + 1)
    put(w, 224, 1)
    put(w, 0, phase)
    return YIELD

def ack_valid(w:mutbuf) -> u64:
    if get(w, 256) != ACTION or equal(w, 256, 384, 64) == 0 or get(w, 320) != get(w, 8) or get(w, 344) != 0:
        return 0
    if get(w, 336) > 4095 or get(w, 328) != 0:
        return 0
    return 1

def close_to(w:mutbuf, slot:u64, phase:u64) -> u64:
    put(w, 216, slot)
    put(w, 208, phase)
    return action(w, 6, slot, 0, 0, 0, 0, 800)

def fail(w:mutbuf, diagnostic:u64, error_number:u64) -> u64:
    put(w, 48, diagnostic)
    put(w, 56, error_number)
    put(w, 120, 0)
    put(w, 0, 900)
    return 0

def own(w:mutbuf, slot:u64, value:u64) -> u64:
    put(w, 128 + (slot - 1) * 8, value)
    return 0

def private_directory(w:mutbuf) -> u64:
    return choose((get(w, 512) & 61440) == 16384 and (get(w, 512) & 4095) == 448 and get(w, 520) == get(w, 528), 1, 0)

def contains_zero(s:readbuf, start:u64, count:u64) -> u64:
    at = 0
    while at < count:
        if byte(s, start + at) == 0:
            return 1
        at = at + 1
    return 0
