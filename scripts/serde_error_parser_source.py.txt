def entry(s:readbuf, o:mutbuf, w:mutbuf) -> u32:
    n = length(s)
    if n < 24 or n > 8388632 or read64(s, 0) != REQUEST:
        return INVALID
    if read64(s, 8) != 1 or read64(s, 16) != n - 24:
        return INVALID
    if length(w) < DATA or n - 24 > length(w) - DATA or length(o) < 64:
        return CAPACITY
    clear(w, P, 256)
    copy_range(w, s, DATA, 24, n - 24)
    pput(w, 0, DATA)
    pput(w, 8, DATA)
    pput(w, 16, DATA + n - 24)
    pput(w, 80, DATA + n - 24)
    push(w, 0)
    result = drive(w)
    if result != 0:
        return result
    return publish(o, w)

def drive(w:mutbuf) -> u64:
    while pget(w, 48) == 0 and pget(w, 184) == 0:
        if pget(w, 176) == 1:
            result = read_value(w)
        else:
            result = advance(w)
        if result != 0:
            return result
    return 0

def read_value(w:mutbuf) -> u64:
    ch = space(w)
    if ch == 256:
        return fail(w, 4, peek_position(w), 0)
    frame = top(w)
    raw = get(w, frame) == 2 and get(w, frame + 8) == 3
    if raw and ch != 34:
        return unexpected(w, ch)
    if ch == 110 or ch == 116 or ch == 102:
        next_byte(w)
        ident(w, ch)
        pput(w, 176, 0)
        return 0
    if ch == 45 or (ch >= 48 and ch <= 57):
        number(w)
        pput(w, 176, 0)
        return 0
    if ch == 34:
        next_byte(w)
        if raw:
            result = string(w, 2)
        else:
            result = string(w, 0)
        if result != 0 or pget(w, 48) != 0:
            return result
        pput(w, 176, 0)
        if raw:
            put(w, frame + 8, 4)
            child = push(w, 3)
            if child == CAPACITY:
                return CAPACITY
            put(w, child + 16, pget(w, 8))
            put(w, child + 24, pget(w, 16))
            put(w, child + 32, pget(w, 0))
            put(w, child + 40, pget(w, 32))
            put(w, child + 48, pget(w, 88))
            pput(w, 8, pget(w, 88))
            pput(w, 16, pget(w, 88) + pget(w, 96))
            pput(w, 0, pget(w, 88))
            pput(w, 32, 0)
            pput(w, 24, pget(w, 24) + 1)
        return 0
    if ch == 91 or ch == 123:
        if pget(w, 32) >= 127:
            return fail(w, 23, peek_position(w), 0)
        next_byte(w)
        kind = 1
        if ch == 123:
            kind = 2
        if push(w, kind) == CAPACITY:
            return CAPACITY
        pput(w, 32, pget(w, 32) + 1)
        pput(w, 176, 0)
        return 0
    return fail(w, 9, peek_position(w), 0)

def advance(w:mutbuf) -> u64:
    frame = top(w)
    kind = get(w, frame)
    stage = get(w, frame + 8)
    if kind == 0 or kind == 3:
        if stage == 0:
            put(w, frame + 8, 1)
            pput(w, 176, 1)
            return 0
        if space(w) != 256:
            return fail(w, 21, peek_position(w), 0)
        if kind == 0:
            pput(w, 184, 1)
            return 0
        pput(w, 8, get(w, frame + 16))
        pput(w, 16, get(w, frame + 24))
        pput(w, 0, get(w, frame + 32))
        pput(w, 32, get(w, frame + 40))
        pput(w, 80, get(w, frame + 48))
        pput(w, 152, 0)
        pput(w, 24, pget(w, 24) - 1)
        pput(w, 40, pget(w, 40) - 1)
        return 0
    if kind == 1:
        return array_next(w, frame, stage)
    return object_next(w, frame, stage)

def array_next(w:mutbuf, frame:u64, stage:u64) -> u64:
    ch = space(w)
    if ch == 256:
        return fail(w, 1, peek_position(w), 0)
    if ch == 93:
        next_byte(w)
        return close_container(w)
    if stage == 0:
        put(w, frame + 8, 1)
        pput(w, 176, 1)
        return 0
    if ch != 44:
        return fail(w, 6, peek_position(w), 0)
    next_byte(w)
    ch = space(w)
    if ch == 93:
        return fail(w, 20, peek_position(w), 0)
    if ch == 256:
        return fail(w, 4, peek_position(w), 0)
    pput(w, 176, 1)
    return 0

def object_next(w:mutbuf, frame:u64, stage:u64) -> u64:
    ch = space(w)
    if ch == 256:
        return fail(w, 2, peek_position(w), 0)
    if ch == 125:
        next_byte(w)
        return close_container(w)
    if stage == 4:
        if ch == 44:
            return fail(w, 20, peek_position(w), 0)
        return fail(w, 21, peek_position(w), 0)
    if stage == 0:
        if ch != 34:
            return fail(w, 16, peek_position(w), 0)
        return object_key(w, frame, 1)
    if ch != 44:
        return fail(w, 7, peek_position(w), 0)
    next_byte(w)
    ch = space(w)
    if ch == 256:
        return fail(w, 4, peek_position(w), 0)
    if ch == 125:
        return fail(w, 20, peek_position(w), 0)
    if ch != 34:
        return fail(w, 16, peek_position(w), 0)
    return object_key(w, frame, 0)

def object_key(w:mutbuf, frame:u64, first:u64) -> u64:
    next_byte(w)
    result = string(w, first)
    if result != 0 or pget(w, 48) != 0:
        return result
    ch = space(w)
    if ch == 256:
        return fail(w, 2, peek_position(w), 0)
    if ch != 58:
        return fail(w, 5, peek_position(w), 0)
    next_byte(w)
    stage = 1
    if first == 1 and pget(w, 104) == 30 and pget(w, 112) == 1:
        stage = 3
    put(w, frame + 8, stage)
    pput(w, 176, 1)
    return 0

def unexpected(w:mutbuf, ch:u64) -> u64:
    code = 0
    if ch == 110 or ch == 116 or ch == 102:
        next_byte(w)
        ident(w, ch)
        if ch == 110:
            code = 100
        else:
            if ch == 116:
                code = 101
            else:
                code = 102
    else:
        if ch == 91:
            code = 103
        else:
            if ch == 123:
                code = 104
            else:
                if ch == 45 or (ch >= 48 and ch <= 57):
                    number(w)
                    if pget(w, 48) != 0:
                        return 0
                    result = number_text(w)
                    if result != 0:
                        return result
                    code = 105
                    if pget(w, 160) == 5:
                        code = 106
                else:
                    return fail(w, 9, peek_position(w), 0)
    if pget(w, 48) != 0:
        return 0
    return fail(w, code, pget(w, 0), 3)

def ident(w:mutbuf, first:u64) -> u64:
    n = 3
    if first == 102:
        n = 4
    at = 0
    while at < n:
        ch = next_byte(w)
        if ch == 256:
            return fail(w, 4, pget(w, 0), 0)
        if ch != ident_byte(first, at):
            return fail(w, 8, pget(w, 0), 0)
        at = at + 1
    return 0

def number(w:mutbuf) -> u64:
    begin = pget(w, 0)
    ch = next_byte(w)
    if ch == 45:
        ch = next_byte(w)
    if ch == 256:
        return fail(w, 4, pget(w, 0), 0)
    if ch < 48 or ch > 57:
        return fail(w, 12, pget(w, 0), 0)
    zero = ch == 48
    ch = peek(w)
    if zero:
        if ch >= 48 and ch <= 57:
            return fail(w, 12, peek_position(w), 0)
    else:
        while ch >= 48 and ch <= 57:
            next_byte(w)
            ch = peek(w)
    if ch == 46:
        next_byte(w)
        mark = pget(w, 0)
        ch = peek(w)
        while ch >= 48 and ch <= 57:
            if ch != 48:
                zero = False
            next_byte(w)
            ch = peek(w)
        if pget(w, 0) == mark:
            if ch == 256:
                return fail(w, 4, peek_position(w), 0)
            return fail(w, 12, peek_position(w), 0)
    if ch == 69 or ch == 101:
        next_byte(w)
        ch = peek(w)
        positive = True
        if ch == 43 or ch == 45:
            positive = ch != 45
            next_byte(w)
        ch = next_byte(w)
        if ch == 256:
            return fail(w, 4, pget(w, 0), 0)
        if ch < 48 or ch > 57:
            return fail(w, 12, pget(w, 0), 0)
        exponent = ch - 48
        overflow = False
        ch = peek(w)
        while ch >= 48 and ch <= 57:
            next_byte(w)
            if overflow == False:
                digit = ch - 48
                if exponent > 214748364 or (exponent == 214748364 and digit > 7):
                    if positive and zero == False:
                        return fail(w, 13, pget(w, 0), 0)
                    overflow = True
                else:
                    exponent = exponent * 10 + digit
            ch = peek(w)
    end = pget(w, 0)
    if pget(w, 152) == 1 and pget(w, 128) - pget(w, 120) == end - begin:
        if same(w, begin, pget(w, 120), end - begin) == 1:
            pput(w, 160, pget(w, 136))
            pput(w, 168, pget(w, 144))
            return 0
    result = classify_private(w, begin, end)
    if result != 0:
        return fail(w, 13, pget(w, 0), 0)
    pput(w, 160, get(w, 7008))
    pput(w, 168, get(w, 7000))
    pput(w, 120, begin)
    pput(w, 128, end)
    pput(w, 136, pget(w, 160))
    pput(w, 144, pget(w, 168))
    pput(w, 152, 1)
    return 0

def number_text(w:mutbuf) -> u64:
    tag = pget(w, 160)
    value = pget(w, 168)
    if tag == 5:
        n = format_bits(w, value)
        if n >= 4294901760 or n > 128:
            return INVALID
        copy_range(w, w, NUMBER_TEXT, 6144, n)
    else:
        at = NUMBER_TEXT
        if tag == 4 and (value >> 63) == 1:
            store(w, at, 45)
            at = at + 1
            value = 0 - value
        n = decimal(w, at, value) - NUMBER_TEXT
    pput(w, 192, n)
    return 0

def push(w:mutbuf, kind:u64) -> u64:
    count = pget(w, 40)
    if count >= 4096:
        return CAPACITY
    frame = STACK + count * 64
    clear(w, frame, 64)
    put(w, frame, kind)
    pput(w, 40, count + 1)
    return frame

def close_container(w:mutbuf) -> u64:
    pput(w, 40, pget(w, 40) - 1)
    pput(w, 32, pget(w, 32) - 1)
    pput(w, 176, 0)
    return 0

def top(w:mutbuf) -> u64:
    return STACK + (pget(w, 40) - 1) * 64

def peek(w:mutbuf) -> u64:
    at = pget(w, 0)
    if at >= pget(w, 16):
        return 256
    return byte(w, at)

def next_byte(w:mutbuf) -> u64:
    value = peek(w)
    if value != 256:
        pput(w, 0, pget(w, 0) + 1)
    return value

def peek_position(w:mutbuf) -> u64:
    at = pget(w, 0)
    if at < pget(w, 16):
        at = at + 1
    return at

def space(w:mutbuf) -> u64:
    value = peek(w)
    while value == 32 or value == 9 or value == 10 or value == 13:
        next_byte(w)
        value = peek(w)
    return value

def fail(w:mutbuf, code:u64, at:u64, category:u64) -> u64:
    if pget(w, 48) != 0:
        return 0
    if category == 0:
        category = 1
        if code <= 4:
            category = 2
    if pget(w, 24) != 0:
        category = 3
    pput(w, 48, code)
    pput(w, 56, at)
    pput(w, 64, pget(w, 8))
    pput(w, 72, category)
    return 0

def same(w:mutbuf, left:u64, right:u64, count:u64) -> u64:
    at = 0
    while at < count:
        if byte(w, left + at) != byte(w, right + at):
            return 0
        at = at + 1
    return 1

def pget(w:mutbuf, at:u64) -> u64:
    return get(w, P + at)

def pput(w:mutbuf, at:u64, value:u64) -> u64:
    put(w, P + at, value)
    return 0

def clear(w:mutbuf, at:u64, count:u64) -> u64:
    end = at + count
    while at < end:
        put(w, at, 0)
        at = at + 8
    return 0
