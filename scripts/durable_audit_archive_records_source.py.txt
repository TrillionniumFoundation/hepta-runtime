def entry(s:readbuf, o:mutbuf, w:mutbuf) -> u32:
    if length(s) < 32 or read64(s, 0) != REQUEST:
        return INVALID
    kind = read64(s, 8)
    raw_size = read64(s, 16)
    if kind < 1 or kind > 3 or read64(s, 24) != 0 or raw_size != length(s) - 32:
        return INVALID
    if length(o) < 32:
        return CAPACITY
    if raw_size > choose(kind == 1, 2048, 65536):
        return refused(o, 2)
    if length(w) < 512 + raw_size * 4:
        return CAPACITY
    fill_range(w, 0, 512, 0)
    put(w, 128, kind)
    at = parse_root(s, w, whitespace(s, 32))
    if at >= CAPACITY or whitespace(s, at) != length(s):
        return refused(o, 1)
    count = get(w, 224)
    required = 128 + count * 16
    if length(o) < required:
        return CAPACITY
    fill_range(o, 0, 128, 0)
    put(o, 0, RESPONSE)
    put(o, 16, kind)
    put(o, 24, required)
    put(o, 32, get(w, 136))
    copy_range(o, w, 40, 144, 64)
    copy_range(o, w, 128, 512, count * 16)
    return required

def refused(o:mutbuf, diagnostic:u64) -> u32:
    fill_range(o, 0, 32, 0)
    put(o, 0, RESPONSE)
    put(o, 8, diagnostic)
    return 32

def parse_root(s:readbuf, w:mutbuf, at:u64) -> u64:
    if at >= length(s):
        return INVALID
    kind = get(w, 128)
    fields = choose(kind == 1, 4, 5)
    if byte(s, at) == 91:
        field = 1
        at = whitespace(s, at + 1)
        while field <= fields:
            at = parse_field(s, w, at, field)
            if at >= CAPACITY:
                return at
            at = whitespace(s, at)
            if at >= length(s):
                return INVALID
            if field < fields:
                if byte(s, at) != 44:
                    return INVALID
                at = whitespace(s, at + 1)
            else:
                if byte(s, at) != 93:
                    return INVALID
                return at + 1
            field = field + 1
        return INVALID
    if byte(s, at) != 123:
        return INVALID
    mask = 0
    at = whitespace(s, at + 1)
    running = 1
    while running == 1:
        if at >= length(s) or byte(s, at) != 34:
            return INVALID
        end = string_info(s, w, at)
        if end >= CAPACITY:
            return end
        first = choose(kind == 1, 1, 5)
        field = identify(s, at + 1, get(w, 24), first, first + fields - 1)
        if field == 0:
            return INVALID
        field = field - first + 1
        flag = 1 << (field - 1)
        if (mask & flag) != 0:
            return INVALID
        mask = mask | flag
        at = whitespace(s, end)
        if at >= length(s) or byte(s, at) != 58:
            return INVALID
        at = parse_field(s, w, whitespace(s, at + 1), field)
        if at >= CAPACITY:
            return at
        at = whitespace(s, at)
        if at >= length(s):
            return INVALID
        if byte(s, at) == 125:
            required = choose(kind == 1, 15, choose(kind == 2, 31, 23))
            if (mask & required) != required:
                return INVALID
            return at + 1
        if byte(s, at) != 44:
            return INVALID
        at = whitespace(s, at + 1)
    return INVALID

def parse_field(s:readbuf, w:mutbuf, at:u64, field:u64) -> u64:
    if at >= length(s):
        return INVALID
    kind = get(w, 128)
    if kind == 1:
        end = unsigned(s, w, at, 64)
        if end >= CAPACITY:
            return end
        put(w, 144 + (field - 1) * 8, get(w, 8))
        return end
    if field == 1:
        end = unsigned(s, w, at, 64)
        if end >= CAPACITY or get(w, 8) > 4294967295:
            return INVALID
        put(w, 136, get(w, 8))
        return end
    if field == 2:
        return identity(s, w, at, 144)
    if field == 3:
        return identity(s, w, at, 160)
    if field == 4:
        if kind == 3 and byte(s, at) == 110:
            return null(s, at)
        end = identity(s, w, at, 184)
        if end >= CAPACITY:
            return end
        put(w, 176, 1)
        return end
    return identities(s, w, at)

def identity(s:readbuf, w:mutbuf, at:u64, offset:u64) -> u64:
    if at >= length(s) or byte(s, at) != 91:
        return INVALID
    at = unsigned(s, w, whitespace(s, at + 1), 64)
    if at >= CAPACITY:
        return at
    put(w, offset, get(w, 8))
    at = whitespace(s, at)
    if at >= length(s) or byte(s, at) != 44:
        return INVALID
    at = unsigned(s, w, whitespace(s, at + 1), 64)
    if at >= CAPACITY:
        return at
    put(w, offset + 8, get(w, 8))
    at = whitespace(s, at)
    if at >= length(s) or byte(s, at) != 93:
        return INVALID
    return at + 1

def identities(s:readbuf, w:mutbuf, at:u64) -> u64:
    if at >= length(s) or byte(s, at) != 91:
        return INVALID
    at = whitespace(s, at + 1)
    if at >= length(s):
        return INVALID
    if byte(s, at) == 93:
        return at + 1
    count = 0
    running = 1
    while running == 1:
        offset = 512 + count * 16
        if offset > length(w) or length(w) - offset < 16:
            return CAPACITY
        at = identity(s, w, at, offset)
        if at >= CAPACITY:
            return at
        count = count + 1
        at = whitespace(s, at)
        if at >= length(s):
            return INVALID
        if byte(s, at) == 93:
            put(w, 224, count)
            put(w, 200, count)
            return at + 1
        if byte(s, at) != 44:
            return INVALID
        at = whitespace(s, at + 1)
    return INVALID

def identify(s:readbuf, start:u64, size:u64, first:u64, last:u64) -> u64:
    code = first
    while code <= last:
        if name_length(code) == size:
            at = start
            i = 0
            equal = 1
            while i < size and equal == 1:
                value = decoded(s, at)
                if (value & 4294967295) != name_byte(code, i):
                    equal = 0
                at = value >> 32
                i = i + 1
            if equal == 1:
                return code
        code = code + 1
    return 0
